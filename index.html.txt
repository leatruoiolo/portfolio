    const scanBtn = document.getElementById('scanBtn');
    const scanResult = document.getElementById('scanResult');
    const urlInput = document.getElementById('urlInput');

    scanBtn.addEventListener('click', () => {
      const value = urlInput.value.trim();

    const gameArea = document.getElementById('gameArea');
    const mouse = document.getElementById('mouse');
    const bug = document.getElementById('bug');
    const startGameBtn = document.getElementById('startGameBtn');
    const resetGameBtn = document.getElementById('resetGameBtn');
    const gameScore = document.getElementById('gameScore');
    const unlockCard = document.getElementById('unlockCard');

    let gameRunning = false;
    let passedBugs = 0;
    let collisionTimer = null;
    let pointTimer = null;

    function jump() {
      if (!gameRunning || mouse.classList.contains('jump')) return;
      mouse.classList.add('jump');
      setTimeout(() => mouse.classList.remove('jump'), 680);
    }

    function startGame() {
      resetGame();
      gameRunning = true;
      startGameBtn.textContent = 'Läuft...';
      bug.classList.add('running');
      gameArea.focus();

      pointTimer = setInterval(() => {
        if (!gameRunning) return;
        passedBugs += 1;
        gameScore.textContent = `${passedBugs} / 3`;

        if (passedBugs >= 3) {
          winGame();
        }
      }, 1650);

      collisionTimer = setInterval(checkCollision, 30);
    }

    function resetGame() {
      gameRunning = false;
      passedBugs = 0;
      gameScore.textContent = '0 / 3';
      startGameBtn.textContent = 'Challenge starten';
      bug.classList.remove('running');
      mouse.classList.remove('jump');
      unlockCard.classList.remove('show');
      clearInterval(collisionTimer);
      clearInterval(pointTimer);
    }

    function winGame() {
      gameRunning = false;
      bug.classList.remove('running');
      startGameBtn.textContent = 'Geschafft ✅';
      unlockCard.classList.add('show');
      clearInterval(collisionTimer);
      clearInterval(pointTimer);
    }

    function loseGame() {
      gameRunning = false;
      bug.classList.remove('running');
      startGameBtn.textContent = 'Nochmal versuchen';
      clearInterval(collisionTimer);
      clearInterval(pointTimer);
      gameScore.textContent = `${passedBugs} / 3 · Bug getroffen`;
    }

    function checkCollision() {
      const mouseRect = mouse.getBoundingClientRect();
      const bugRect = bug.getBoundingClientRect();
      const overlap = !(
        mouseRect.right < bugRect.left + 8 ||
        mouseRect.left > bugRect.right - 8 ||
        mouseRect.bottom < bugRect.top + 8 ||
        mouseRect.top > bugRect.bottom - 8
      );

      if (overlap) {
        loseGame();
      }
    }

    startGameBtn.addEventListener('click', startGame);
    resetGameBtn.addEventListener('click', resetGame);
    gameArea.addEventListener('click', jump);
    document.addEventListener('keydown', (event) => {
      if (event.code === 'Space') {
        event.preventDefault();
        jump();
      }
    });
